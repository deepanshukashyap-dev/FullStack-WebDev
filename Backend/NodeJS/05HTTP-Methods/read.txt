//Creating server
const myServer = http.createServer((req,res) => {
    //req: for client side all data , res: for to send the response to the server
    console.log("New Request received! "); //ignore
    const log = `${Date.now()}: ${req.method} ${req.url} New Req is received\n`; //Changes is"${req.method}"
    const myUrl = url.parse(req.url,true); //converting URL string into url obj , true karne se query bhi mil jata hai
    console.log(myUrl); //ignore

    fs.appendFile("log.txt", log, (err)=>{
        if(err){
            console.log("Error!: ",err);
        }else{
            console.log("log is successfully append to file log.txt(Asynchronously)");
            switch(myUrl.pathname){
                case '/': 
                    res.end("Home Page: Response from the server");
                    break;
                case '/about': 
                    res.end("About Page");
                    break;
                default:
                    res.end("404");

            }
        }
    })
});
